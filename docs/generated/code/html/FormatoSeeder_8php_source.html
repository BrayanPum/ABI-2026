126 private function campos(FormatoTipo $tipo, array $campos): void
127 {
128 foreach ($campos as $orden => $data) {
129 FormatoCampo::create([
130 'formato_tipo_id' => $tipo->id,
131 'nombre' => \Illuminate\Support\Str::slug($data['etiqueta'], '_'),
132 'etiqueta' => $data['etiqueta'],
133 'tipo' => $data['tipo'],
134 'opciones' => $data['opciones'] ?? null,
135 'requerido' => $data['requerido'],
136 'seccion' => $data['seccion'],
137 'orden' => $orden,
138 ]);
139 }
140 }
