38 public function index(Request $request): JsonResponse
39 {
40 try {
41 // Validar parámetros de entrada
42 $validated = $request->validate([
43 'per_page' => 'nullable|integer|min:1|max:100',
44 'search' => 'nullable|string|max:255',
45 'roles' => 'nullable', // Puede ser string o array
46 ]);
47
48 // Configurar paginación con valor por defecto y límite máximo
49 $perPage = (int) ($validated['per_page'] ?? 15);
50 $perPage = $perPage > 0 ? min($perPage, 100) : 15;
51
52 // Iniciar consulta base
53 $query = ResearchStaffContent::query()->withCount('contentVersions');
54
55 // Aplicar filtro de búsqueda si existe
56 if (!empty($validated['search'])) {
57 $search = trim($validated['search']);
58 $query->where(function ($q) use ($search) {
59 $q->where('name', 'like', '%' . $search . '%')
60 ->orWhere('description', 'like', '%' . $search . '%');
61
62 // Si el término de búsqueda es numérico, buscar también por ID
63 if (is_numeric($search)) {
64 $q->orWhere('id', (int)$search);
65 }
66 });
67 }
68
69 // Aplicar filtro de roles si existe
70 $rolesFilter = $request->query('roles');
71 if ($rolesFilter !== null && $rolesFilter !== '') {
72 // Normalizar el filtro de roles a array
73 if (is_string($rolesFilter)) {
74 $rolesFilter = json_decode($rolesFilter, true);
75 if (json_last_error() !== JSON_ERROR_NONE || !is_array($rolesFilter)) {
76 $rolesFilter = array_filter(array_map('trim', explode(',', $rolesFilter)));
77 }
78 }
79
80 if (!is_array($rolesFilter)) {
81 $rolesFilter = [$rolesFilter];
82 }
83
84 // Aplicar filtro JSON para cada rol especificado
85 foreach ($rolesFilter as $role) {
86 $role = trim((string) $role);
87 if ($role !== '') {
88 $query->whereJsonContains('roles', $role);
89 }
90 }
91 }
92
93 // Obtener resultados paginados ordenados alfabéticamente
94 $contents = $query
95 ->orderBy('name')
96 ->paginate($perPage)
97 ->withQueryString();
98
99 return response()->json($contents);
100
101 } catch (ValidationException $e) {
102 return response()->json([
103 'message' => 'Los datos de entrada no son válidos.',
104 'errors' => $e->errors(),
105 ], 422);
106 } catch (\Exception $e) {
107 Log::error('Error al listar contenidos: ' . $e->getMessage(), [
108 'trace' => $e->getTraceAsString()
109 ]);
110
111 return response()->json([
112 'message' => 'Ocurrió un error al obtener los contenidos.',
113 ], 500);

173 public function show(ResearchStaffContent $content): JsonResponse
174 {

183 // Attempt to load the related versions but fail gracefully if the connection does not expose them

186 $query->orderByDesc('content_version.created_at');

188 } catch (QueryException $exception) {

190 'content_id' => $content->id,
191 'error' => $exception->getMessage(),

193 }
194
195 $response = [
196 'id' => $content->id,
197 'name' => $content->name,
198 'description' => $content->description,
199 'roles' => Arr::wrap($content->roles),
200 'created_at' => $content->created_at,
201 'updated_at' => $content->updated_at,
202 ];
203
204 if ($content->relationLoaded('versions')) {
205 $response['versions'] = $content->versions->map(static function ($version) {
206 return [
207 'id' => $version->id,
208 'name' => $version->name,
209 'pivot' => [
210 'id' => $version->pivot->id ?? null,
211 'value' => $version->pivot->value ?? null,
212 'created_at' => $version->pivot->created_at ?? null,
213 'updated_at' => $version->pivot->updated_at ?? null,
214 ],
215 ];
216 });
217 }
218
219 return response()->json($response);
