30 public function index(Request $request): View
31 {
32 try {
33 // Obtener parámetros de filtrado
34 $search = $request->get('search');
35 $researchGroupId = $request->get('research_group_id');
36 $perPage = (int) $request->get('per_page', 10);
37 $perPage = $perPage > 0 ? min($perPage, 100) : 10;
38
39 // Query base con relaciones y contadores
40 $investigationLines = ResearchStaffInvestigationLine::query()
41 ->with(['researchGroup'])
42 ->withCount('thematicAreas')
43 ->when($search, function ($query, string $search) {
44 $query->where(function ($q) use ($search) {
45 $q->where('name', 'like', "%{$search}%")
46 ->orWhere('description', 'like', "%{$search}%");
47 });
48 })
49 ->when($researchGroupId, function ($query, $groupId) {
50 $query->where('research_group_id', $groupId);
51 })
52 ->orderBy('name')
53 ->paginate($perPage)
54 ->appends($request->query());
55
56 // Obtener grupos de investigación para filtro
57 $researchGroups = ResearchStaffResearchGroup::orderBy('name')->pluck('name', 'id');
58
59 return view('investigation-lines.index', [
60 'investigationLines' => $investigationLines,
61 'researchGroups' => $researchGroups,
62 'search' => $search,
63 'researchGroupId' => $researchGroupId,
64 'perPage' => $perPage,
65 ]);
66
67 } catch (\Exception $e) {
68 Log::error('Error al listar líneas de investigación: ' . $e->getMessage());
69
70 return view('investigation-lines.index', [
71 'investigationLines' => collect(),
72 'researchGroups' => collect(),
73 'search' => '',
74 'researchGroupId' => null,
75 'perPage' => 10,
76 'error' => 'Ocurrió un error al cargar las líneas de investigación.',
77 ]);
