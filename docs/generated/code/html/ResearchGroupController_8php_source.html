29 public function index(Request $request): View
30 {
31 try {
32 // Obtener parámetros de filtrado
33 $search = $request->get('search');
34 $perPage = (int) $request->get('per_page', 10);
35 $perPage = $perPage > 0 ? min($perPage, 100) : 10;
36
37 // Query base con contadores de relaciones
38 $researchGroups = ResearchStaffResearchGroup::query()
39 ->withCount(['programs', 'investigationLines'])
40 ->when($search, function ($query, string $search) {
41 $query->where(function ($q) use ($search) {
42 $q->where('name', 'like', "%{$search}%")
43 ->orWhere('initials', 'like', "%{$search}%");
44 });
45 })
46 ->orderBy('name')
47 ->paginate($perPage)
48 ->appends($request->query());
49
50 return view('research-groups.index', [
51 'researchGroups' => $researchGroups,
52 'search' => $search,
53 'perPage' => $perPage,
54 ]);
55
56 } catch (\Exception $e) {
57 Log::error('Error al listar grupos de investigación: ' . $e->getMessage());
58
59 return view('research-groups.index', [
60 'researchGroups' => collect(),
61 'search' => '',
62 'perPage' => 10,
63 'error' => 'Ocurrió un error al cargar los grupos de investigación.',
64 ]);
