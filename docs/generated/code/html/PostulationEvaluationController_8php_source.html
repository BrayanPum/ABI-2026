48 public function evaluate(Request $request, Postulation $postulation)
49 {
50 $user = Auth::user();
51 $professor = Professor::where('user_id', $user->id)->firstOrFail();
52
53 // Validar permisos de city_program
54 if ($postulation->leadStudent->city_program_id !== $professor->city_program_id) {
55 abort(403, 'No tienes permiso para evaluar esta postulación.');
56 }
57
58 $request->validate([
59 'status' => 'required|in:approved,rejected',
60 'review_comment' => 'required|string|max:1000',
61 ]);
62
63 if ($postulation->status !== 'pending') {
64 return back()->with('error', 'Esta postulación ya ha sido evaluada.');
65 }
66
67 DB::beginTransaction();
68 try {
69 $postulation->update([
70 'status' => $request->status,
71 'review_comment' => $request->review_comment,
72 ]);
73
74 // Si se rechaza, liberar la prioridad
75 if ($request->status === 'rejected') {
76 $this->cleanupPriority($postulation);
77 }
78
79 if ($request->status === 'approved') {
80 $project = $postulation->project;
81 $activePeriod = AcademicCalendarService::currentActivePeriodOrFail();
82 $assignedStatusId = ProjectStatus::where('name', 'Asignado')->firstOrFail()->id;
83
84 // 1. Asignar proyecto y vincular con la postulación aprobada
85 $project->update([
86 'project_status_id' => $assignedStatusId,
87 'approved_postulation_id' => $postulation->postulation_id,
88 'assignment_academic_period_id' => $activePeriod->id,
89 'assigned_at' => now(),
90 ]);
91
92 // 2. Sincronizar estudiantes con la referencia de la postulación
93 $studentIds = $postulation->members->pluck('student_id')->toArray();
94
95 // Usamos sync con datos adicionales para el pivot si es necesario,
96 // o simplemente adjuntamos la postulation_id
97 $syncData = [];
98 foreach ($studentIds as $id) {
99 $syncData[$id] = ['postulation_id' => $postulation->postulation_id];
100 }
101 $project->students()->sync($syncData);
102
103 // 3. Cancelar otras postulaciones para TODOS los integrantes involucrados
104 foreach ($studentIds as $studentId) {
105 $otherPostulations = Postulation::where('status', 'pending')
106 ->where('postulation_id', '!=', $postulation->postulation_id)
107 ->whereHas('members', function ($q) use ($studentId) {
108 $q->where('student_id', $studentId);
109 })
110 ->get();
111
112 foreach ($otherPostulations as $other) {
113 $other->update([
114 'status' => 'rejected',
115 'review_comment' => 'Cancelada automáticamente por aprobación de otra postulación.',
116 ]);
117 // Liberar prioridades de las postulaciones canceladas
118 $this->cleanupPriority($other);
119 }
120 }
121
122 AcademicCalendarService::recordProjectStage(
123 $project,
124 'assigned',
125 $activePeriod,
126 Auth::id(),
127 'Proyecto asignado vía aprobación de postulación. '.$request->review_comment,
128 ['postulation_id' => $postulation->postulation_id, 'student_ids' => $studentIds]
129 );
130 }
131
132 DB::commit();

135 ->with('success', 'Postulación evaluada correctamente.');

147 private function cleanupPriority(Postulation $postulation): void
148 {
149 $priorityRecord = PostulationPriority::where('postulation_id', $postulation->postulation_id)
150 ->first();
151
152 if ($priorityRecord) {
153 $studentId = $priorityRecord->student_id;
154 $order = $priorityRecord->priority_order;
155
156 // Eliminar el registro de prioridad actual
157 $priorityRecord->delete();
158
159 // Reordenar las prioridades superiores del mismo estudiante para cerrar el hueco
160 PostulationPriority::where('student_id', $studentId)
161 ->where('priority_order', '>', $order)
162 ->decrement('priority_order');
