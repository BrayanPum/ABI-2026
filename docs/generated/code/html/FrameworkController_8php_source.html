30 public function index(Request $request): View
31 {
32 // Obtener parámetros de filtrado y búsqueda
33 $search = $request->get('search');
34 $year = $request->get('year');
35 $perPageOptions = [10, 20, 30];
36 $perPage = (int) $request->get('per_page', $perPageOptions[0]);
37
38 // Validar que per_page esté en las opciones permitidas
39 if (! in_array($perPage, $perPageOptions, true)) {
40 $perPage = $perPageOptions[0];
41 }
42
43 try {
44
45 // Query base - solo frameworks no eliminados
46 $query = ResearchStaffFramework::query()->withCount('contentFrameworks');
47
48 // Aplicar filtro de búsqueda
49 if ($search) {
50 $query->where(function ($q) use ($search) {
51 $q->where('name', 'like', '%' . $search . '%')
52 ->orWhere('description', 'like', '%' . $search . '%');
53
54 // Si la búsqueda es numérica, también buscar por ID
55 if (is_numeric($search)) {
56 $q->orWhere('id', $search);
57 }
58 });
59 }
60
61 // Aplicar filtro por año
62 if ($year) {
63 $query->where(function ($q) use ($year) {
64 $q->where('start_year', '<=', $year)
65 ->where(function ($subQ) use ($year) {
66 $subQ->whereNull('end_year')
67 ->orWhere('end_year', '>=', $year);
68 });
69 });
70 }
71
72 // Ordenamiento por defecto
73 $query->orderBy('start_year', 'desc')
74 ->orderBy('name', 'asc');
75
76 // Paginación
77 $frameworks = $query->paginate($perPage);
78
79 // Mantener parámetros en paginación
80 $frameworks->appends($request->query());
81
82 return view('framework.index', [
83 'frameworks' => $frameworks,
84 'search' => $search,
85 'year' => $year,
86 'perPage' => $perPage,
87 'perPageOptions' => $perPageOptions,
88 'i' => ($frameworks->currentPage() - 1) * $frameworks->perPage(),
89 ]);
90
91 } catch (\Exception $e) {
92 Log::error('Error al listar frameworks: ' . $e->getMessage());
93
94 $emptyPaginator = new LengthAwarePaginator(
95 collect(),
96 0,
97 $perPage,
98 1,
99 [
100 'path' => $request->url(),
101 'query' => $request->query(),
102 ]
103 );
104
105 return view('framework.index', [
106 'frameworks' => $emptyPaginator,
107 'search' => $search,
108 'year' => $year,
109 'perPage' => $perPage,
110 'perPageOptions' => $perPageOptions,
111 'i' => 0,
112 'error' => 'Ocurrió un error al cargar los marcos.',
113 ]);
114 }
