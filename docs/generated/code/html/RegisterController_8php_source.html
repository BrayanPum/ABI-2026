218 public function register(Request $request)
219 {
220 // Validate the data
221 $this->validator($request->all())->validate();
222
223 // Create the user (without logging in)
224 $user = $this->create($request->all());
225
226 // Fire registration event (optional, for consistency)
227 event(new Registered($user));
228
229 // Fire custom UserCreated event for our notification system
230 event(new UserCreated($user, $request->only(['name', 'last_name'])));
231
232 // Handle post-registration logic and redirection
233 if ($response = $this->registered($request, $user)) {
234 return $response;
235 }
236
237 // Fallback (should not execute if registered() returns a response)
238 return $request->wantsJson()
239 ? new JsonResponse([], 201)
240 : redirect($this->redirectPath());
241 }
