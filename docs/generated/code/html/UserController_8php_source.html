254 public function update(\Illuminate\Http\Request $request, ResearchStaffUser $user): RedirectResponse
255 {
256 // Validate common fields
257 $validated = $request->validate([
258 'email' => [
259 'required',
260 'email',
261 Rule::unique('users')->ignore($user->id)
262 ],
263 'state' => 'required|in:1,0',
264 'role' => 'required|in:student,professor,committee_leader,research_staff',
265 'password' => 'nullable|min:8|confirmed',
266 ]);
267
268 $newRole = $request->role;
269 $oldRole = $user->role;
270
271 $additional = $this->validateRoleFields($request, $newRole);
272
273 // Update the user
274 $user->update([
275 'email' => $validated['email'],
276 'state' => $validated['state'],
277 'role' => $newRole,
278 'password' => $validated['password'] ? Hash::make($validated['password']) : $user->password,
279 ]);
280
281
282 $isTeacherRoleChange =
283 ($oldRole === 'professor' && $newRole === 'committee_leader') ||
284 ($oldRole === 'committee_leader' && $newRole === 'professor');
285
286 // If there is a role change, we need to manage the information from the role tables
287
288 if ($newRole !== $oldRole && !$isTeacherRoleChange) {
289 // Delete (soft delete) the record from the previous table
290 $this->deleteOldRoleRecord($user, $oldRole);
291
292 // Create a new record in the new role's table
293 $this->createNewRoleRecord($user, $newRole, $additional);
294 }
295 // If the role does not change, we update the existing record
296 else {
297 $this->updateExistingRoleRecord($user, $newRole, $additional);
298 }
299
300 return redirect()
301 ->route('users.index')
302 ->with('success', "Usuario '{$user->email}' actualizado correctamente.");
